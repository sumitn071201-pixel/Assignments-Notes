Q3. Create a UDF tax_amount (income DECIMAL(10,2)) functon that returns based on slabs:

5% for 2.5-5L.
20% off 5-10L.
30% more than 10L.

Drop function if exists tax_amount;
delimiter $
create function tax_amount(income DECIMAL(10,2)) RETURNS INT
deterministic
begin
    declare Income, Taxamt FLOAT;
    lbl:loop
        if Income > 250000 & Income < 500000 then
	Taxamt = Income*0.05;
	
	else if Income > 500000 & Income <1000000 then
	Taxamt = Income*0.10;

	else if Income > 1000000 then
	Taxamt = Income*0.30;
          leave lbl;
       end if;
       
    end loop lbl;
    return(Taxamt);
end $
delimiter ;
